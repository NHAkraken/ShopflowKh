Supabase Multi-Tenant Connection

Production PostgreSQL backend with Row Level Security

PostgreSQL Multi-Tenant Schema & Atomic RPC

Copy this directly into Supabase SQL Editor to initialize tables, RLS policies, and stored procedures.

-- ==============================================================
-- SHOPFLOW KH: Multi-Tenant Enterprise POS Schema & Atomic RPC
-- ==============================================================

-- 1. Create Shops Table
create table if not exists public.shops (
  id uuid primary key default gen_random_uuid(),
  name text not null,
  address text,
  currency text default 'USD',
  exchange_rate numeric default 4100,
  bakong_id text default 'shopflow@aclb',
  merchant_city text default 'Phnom Penh',
  created_at timestamptz default now()
);

-- 2. Profiles Table with RBAC (owner / cashier)
create table if not exists public.profiles (
  id uuid primary key references auth.users on delete cascade,
  shop_id uuid references public.shops(id) on delete cascade,
  email text,
  role text default 'cashier' check (role in ('owner', 'cashier')),
  created_at timestamptz default now()
);

-- 3. Products Table
create table if not exists public.products (
  id uuid primary key default gen_random_uuid(),
  shop_id uuid not null references public.shops(id) on delete cascade,
  name text not null,
  sku text,
  barcode text not null,
  category text default 'General',
  cost_price numeric default 0,
  selling_price numeric not null,
  stock_in_hand integer default 0,
  opening_stock integer default 0,
  min_stock_alert integer default 5,
  created_at timestamptz default now()
);

-- 4. Sales Master Table
create table if not exists public.sales (
  id uuid primary key default gen_random_uuid(),
  shop_id uuid not null references public.shops(id) on delete cascade,
  receipt_code text not null,
  cashier_id uuid references auth.users,
  payment_method text default 'cash', -- 'cash', 'khqr'
  total_usd numeric not null,
  total_khr numeric not null,
  tendered_usd numeric,
  change_usd numeric,
  created_at timestamptz default now()
);

-- 5. Sale Line Items Table
create table if not exists public.sale_items (
  id uuid primary key default gen_random_uuid(),
  sale_id uuid not null references public.sales(id) on delete cascade,
  product_id uuid references public.products(id),
  product_name text not null,
  quantity integer not null,
  unit_price numeric not null,
  cost_price numeric default 0,
  line_total numeric not null
);

-- 6. Stock Movements Audit Ledger
create table if not exists public.stock_movements (
  id uuid primary key default gen_random_uuid(),
  shop_id uuid not null references public.shops(id) on delete cascade,
  product_id uuid references public.products(id),
  movement_type text not null, -- 'sale', 'receipt', 'adjustment', 'closing'
  quantity_change integer not null,
  notes text,
  created_at timestamptz default now()
);

-- 7. Daily Closing Records
create table if not exists public.closing_records (
  id uuid primary key default gen_random_uuid(),
  shop_id uuid not null references public.shops(id) on delete cascade,
  closing_date date default current_date,
  total_revenue_usd numeric not null,
  total_revenue_khr numeric not null,
  total_profit_usd numeric default 0,
  transactions_count integer default 0,
  closed_by uuid references auth.users,
  created_at timestamptz default now()
);

-- ==============================================================
-- 8. Row Level Security (RLS) Isolation
-- ==============================================================
alter table public.shops enable row level security;
alter table public.profiles enable row level security;
alter table public.products enable row level security;
alter table public.sales enable row level security;
alter table public.sale_items enable row level security;
alter table public.stock_movements enable row level security;
alter table public.closing_records enable row level security;

create or replace function public.current_user_shop_id()
returns uuid language sql stable as $$
  select shop_id from public.profiles where id = auth.uid()
$$;

create policy "Users can view their shop" on public.shops
  for select using (id = public.current_user_shop_id());

create policy "Users can view profile" on public.profiles
  for select using (auth.uid() = id);

create policy "Shop members access products" on public.products
  for all using (shop_id = public.current_user_shop_id());

create policy "Shop members access sales" on public.sales
  for all using (shop_id = public.current_user_shop_id());

create policy "Shop members access sale items" on public.sale_items
  for all using (sale_id in (select id from public.sales where shop_id = public.current_user_shop_id()));

create policy "Shop members access movements" on public.stock_movements
  for all using (shop_id = public.current_user_shop_id());

create policy "Shop members access closing" on public.closing_records
  for all using (shop_id = public.current_user_shop_id());

-- ==============================================================
-- 9. Atomic Process Sale RPC Function (Prevents Race Conditions)
-- ==============================================================
create or replace function public.process_sale(
  p_shop_id uuid,
  p_receipt_code text,
  p_payment_method text,
  p_total_usd numeric,
  p_total_khr numeric,
  p_tendered_usd numeric,
  p_change_usd numeric,
  p_items jsonb
)
returns uuid
language plpgsql
security definer
as $$
declare
  v_sale_id uuid;
  v_item jsonb;
  v_prod_id uuid;
  v_qty int;
  v_price numeric;
  v_cost numeric;
  v_name text;
begin
  -- Insert Master Sale Record
  insert into public.sales (
    shop_id, cashier_id, receipt_code, payment_method, 
    total_usd, total_khr, tendered_usd, change_usd
  ) values (
    p_shop_id, auth.uid(), p_receipt_code, p_payment_method,
    p_total_usd, p_total_khr, p_tendered_usd, p_change_usd
  ) returning id into v_sale_id;

  -- Atomic Line Items & Inventory Decrements
  for v_item in select * from jsonb_array_elements(p_items)
  loop
    v_prod_id := (v_item->>'product_id')::uuid;
    v_qty := (v_item->>'quantity')::int;
    v_price := (v_item->>'unit_price')::numeric;
    v_cost := coalesce((v_item->>'cost_price')::numeric, 0);
    v_name := v_item->>'product_name';

    insert into public.sale_items (
      sale_id, product_id, product_name, quantity, unit_price, cost_price, line_total
    ) values (
      v_sale_id, v_prod_id, v_name, v_qty, v_price, v_cost, (v_qty * v_price)
    );

    update public.products
    set stock_in_hand = greatest(0, stock_in_hand - v_qty)
    where id = v_prod_id and shop_id = p_shop_id;

    insert into public.stock_movements (
      shop_id, product_id, movement_type, quantity_change, notes
    ) values (
      p_shop_id, v_prod_id, 'sale', -v_qty, 'Sale receipt #' || p_receipt_code
    );
  end loop;

  return v_sale_id;
end;
$$;
